* LSP
** Lsp Bridge
#+begin_src emacs-lisp :tangle "init.el"
  (use-package lsp-bridge
    :disabled
    :hook (acm-mode . colorful-mode)
    :ensure
    (:host github
  	 :repo "manateelazycat/lsp-bridge"
  	 :files (:defaults "*.el" "*.py" "acm" "core" "langserver" "multiserver" "resources")
  	 :build (:not compile))
    :bind
    (:map acm-mode-map
  	 ("<tab>" . 'acm-doc-toggle)
  	 ("TAB"   . 'acm-doc-toggle)
  	 ("C-M-n" . 'acm-doc-scroll-up)
  	 ("C-M-p" . 'acm-doc-scroll-down)
  	 ("C-n"   . 'acm-select-next-page)
  	 ("C-p"   . 'acm-select-prev-page)
  	 ("C-j"   . 'acm-select-next)
  	 ("C-k"   . 'acm-select-prev))
    :config
    (advice-add
     'completion-at-point :override #'lsp-bridge-popup-complete-menu)
    (set-face-attribute 'acm-frame-default-face nil
  		      :background (face-attribute 'fringe :background))
    (set-face-attribute 'acm-frame-select-face nil
  		      :background (face-attribute 'vertico-current :background)
  		      :foreground "white")
    :custom
    ;; ACM settings
    (acm-doc-frame-boundary 'display)
    (acm-candidate-match-function 'orderless-flex)
    (acm-backend-lsp-match-mode "substring")
    (acm-backend-lsp-candidates-max-number 200)
    (tab-always-indent 'complete)
    (lsp-bridge-user-multiserver-dir (concat user-emacs-directory "components/multiserver"))
    (lsp-bridge-user-langserver-dir (concat user-emacs-directory "components/langserver"))
    (lsp-bridge-completion-in-string-file-types '("vue" "dart" "html" "tsx"))
    (lsp-bridge-multi-lang-server-extension-list
     '((("tsx") . "tsx_tailwindcss")))
    (lsp-bridge-multi-lang-server-mode-list
     '(((tsx-ts-mode) . "tsx_tailwindcss")))
    (lsp-bridge-get-language-id
     (lambda (project-path file-path server-name extension-name)
       (when (string-equal server-name "tailwindcss")
         (pcase extension-name
           ("tsx" "typescriptreact")
           ("jsx" "javascriptreact")
           ("ts"  "typescript")
           ("js"  "javascript")
           (_ extension-name))))))
#+end_src

** LSP-UI
#+begin_src emacs-lisp :tangle "init.el"
  (use-package lsp-ui
    :disabled
    :if (display-graphic-p)
    :after lsp-mode
    :config
    (set-face-attribute  'lsp-ui-sideline-global nil;; иначе уёбищный шрифт будет
  		       :family "Iosevka NF" )
    (set-face-attribute 'lsp-ui-doc-background nil
  		      :background (face-attribute 'default :background nil t))
    (set-face-attribute 'lsp-ui-doc-highlight-hover nil
  		      :background (face-attribute 'default :background nil t))
    :custom
    ;; Sideline
    (lsp-ui-doc-border (face-background 'isearch))
    (lsp-ui-sideline-enable t)
    (lsp-ui-sideline-show-diagnostics nil)
    (lsp-ui-sideline-show-hover nil)
    (lsp-ui-sideline-show-code-actions nil)
    (lsp-ui-sideline-delay 0.2)
    ;; Doc popup
    (lsp-ui-doc-enable t)
    (lsp-ui-doc-position 'at-point)
    (lsp-ui-doc-show-with-cursor nil) ;;after command
    (lsp-ui-doc-show-with-moust t)
    (lsp-ui-doc-max-height 20)
    :bind
    (:map lsp-ui-mode-map
              ("M-."     . 'lsp-ui-peek-find-definitions)
              ("M-?"     . 'lsp-ui-peek-find-references)
              ("C-c l d" . 'lsp-ui-doc-glance)
              ("C-c l i" . 'lsp-ui-imenu)))
#+end_src
